Service is not probably not defined in any compose.yaml file labels for Traefik.

You should have something like this on your project:

        
services:
app:
    container_name: "${COMPOSE_PROJECT_NAME}-app"
    image: someimage
    labels:
    - "traefik.enable=true"
    - "traefik.http.routers.${COMPOSE_PROJECT_NAME}-app.entrypoints=https"
    - "traefik.http.routers.${COMPOSE_PROJECT_NAME}-app.rule=Host(`[HOSTNAME]`)"
    - "traefik.http.routers.${COMPOSE_PROJECT_NAME}-app.tls=true"
    - "traefik.http.services.${COMPOSE_PROJECT_NAME}-app.loadbalancer.server.port=8080"
    - "traefik.docker.network=${PREFIX}-network"
    networks:
    - default
    - ${PREFIX}-network

networks:
${PREFIX}-network:
    external: true
